				if (!gl.getShaderParameter(shader, gl.COMPILE_STATUS)) throw new Error(gl.getShaderInfoLog(shader));

				// A tiny overscan prevents a soft-focus setting from revealing the canvas edge.
				canvas.style.transform = currentSoftFocus > 0.001 ? `scale(${1 + currentSoftFocus * 0.0025})` : 'none';
			};
			updatePresentation();

				if (!(control instanceof HTMLInputElement) || control.dataset.shaderControl !== shaderId) return;
				const key = control.dataset.param;
				if (!key) return;
				if (key === 'darkness') {
					document.querySelectorAll(`[data-shader-darkness-overlay="${shaderId}"]`).forEach((overlay) => {

					});

				});

					control.value = key === 'darkness' ? '0' : String(resetValues[key]);
					control.dispatchEvent(new Event('input', { bubbles: true }));
				});
				document.dispatchEvent(new CustomEvent('model-shader-preset', { detail: { shaderId, params: resetValues } }));
			});

			const resize = () => {
				if (!sizeDirty) return;
				const bounds = canvas.getBoundingClientRect();
				const width = Math.max(1, Math.round(bounds.width * currentResolutionScale));
				const height = Math.max(1, Math.round(bounds.height * currentResolutionScale));
				if (canvas.width !== width || canvas.height !== height) {
					canvas.width = width;
					canvas.height = height;
					gl.viewport(0, 0, width, height);
				}
				sizeDirty = false;
			};
			const render = (now) => {
				if (now - lastFrame >= 1000 / maxFps || reducedMotion) {
					resize();
					gl.uniform2f(resolution, canvas.width, canvas.height);
					if (layoutResolution) gl.uniform2f(layoutResolution, canvas.width / canvasCoverage, canvas.height);
					if (viewport) gl.uniform2f(viewport, 1 - canvasCoverage, canvasCoverage);
					if (time) gl.uniform1f(time, startTime + (reducedMotion ? 0 : (now - startedAt) / 1000 * values.speed));
					Object.entries(locations).forEach(([key, location]) => {
						if (location) gl.uniform1f(location, values[key]);
					});
					gl.drawArrays(gl.TRIANGLES, 0, 3);
					canvas.dataset.ready = 'true';
					lastFrame = now;
				}
				if (!reducedMotion && visible) frame = requestAnimationFrame(render);
			};

			new ResizeObserver(() => { sizeDirty = true; }).observe(canvas);
			new IntersectionObserver(([entry]) => {
				visible = entry.isIntersecting;
				cancelAnimationFrame(frame);
				if (visible) frame = requestAnimationFrame(render);
				}).observe(canvas);
			}
		};

		// prototype contains many shader families, and eagerly allocating every
		// context can evict the selected hero before its controls are used.

			if (!entry.isIntersecting) return;
			activationObserver.disconnect();
			initialize();
		}, { rootMargin: '160px' });
		activationObserver.observe(canvas);
	}

<p><a href="https://huggingface.co/LiquidAI/d1-omni-600M">d1-omni-600M</a> is a compact zero-shot classification model from <a href="https://www.liquid.ai/blog/d1-open">Liquid AI</a>, built on LFM2.5-Encoder-350M with vision and audio encoders. Given text, images, or a voice clip and a set of questions, it returns structured answers in a single forward pass.</p>
<p>Describe your task and candidate labels in natural language to route support tickets, classify voice commands, moderate content, or inspect images. Changing tasks requires changing the questions and labels, without task-specific fine-tuning.</p>
<p>The model answers three types of questions: <code>noul</code> returns a probability for a yes/no question, <code>choice</code> selects from named options, and <code>score</code> returns a rating across ordered levels. A request can include text with images or text with audio. Images and audio cannot be combined in the same request.</p>
<p>This guide uses FP16 to run text, image, audio, and batch examples on the GPU. For the larger model and its reported Jetson latency, see <a href="/models/d1-3b/#model-details">d1-3B on Jetson</a>.</p>
